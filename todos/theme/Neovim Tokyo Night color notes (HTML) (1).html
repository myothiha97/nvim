Override config

require("tokyonight").setup({
  on_highlights = function(hl)
    hl["@variable.member"] = { fg = "#7AB8AE" }
    hl["@property"] = { fg = "#7AB8AE" }
    hl["@string"] = { fg = "#95B276" }
  end,
})
vim.cmd.colorscheme("tokyonight")

If :Inspect shows a different highlight group on properties or strings, use that group name instead.
